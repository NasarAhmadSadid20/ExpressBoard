app.post("/register", async (req, res) => {
    const { username, email, phone, password } = req.body;

    // بررسی قدرت رمز عبور
    const passwordStrength = zxcvbn(password);
    if (passwordStrength.score < 3) {
        return res.status(400).json({
            message: "رمز عبور ضعیف است. لطفاً رمز قوی‌تری انتخاب کنید.",
            suggestions: passwordStrength.feedback.suggestions
        });
    }

    // بررسی تکراری نبودن کاربر
    const foundUser = await User.findOne({ $or: [{ email }, { phone }] });
    if (foundUser) {
        return res.status(400).json({ message: "کاربر از قبل وجود دارد." });
    }

    // هش کردن و ذخیره رمز عبور
    const hashedPassword = await cryptPassword(password);
    const newUser = new User({ username, email, phone, password: hashedPassword });
    await newUser.save();

    res.status(201).json({ message: "ثبت‌نام موفقیت‌آمیز بود!" });
});

registerUser(req, res) {
    const { name, email, password } = req.body;
    
    db.User.findOne({ email: email })
        .then(existingUser => {
            if (existingUser) {
                return res.render("register", { error: "این ایمیل قبلاً ثبت شده است." });
            }

            db.User.create({ name, email, password })
                .then(user => {
                    res.redirect("/login"); // موفقیت‌آمیز بود، به صفحه لاگین هدایت شود
                })
                .catch(err => {
                    let errorMessage = "خطایی رخ داده است.";
                    if (err.errors && err.errors.password) {
                        errorMessage = err.errors.password.message; // پیام ارور رمز عبور
                    }
                    res.render("register", { error: errorMessage });
                });
        })
        .catch(err => {
            res.render("register", { error: "مشکلی در سرور رخ داده است." });
        });
}
